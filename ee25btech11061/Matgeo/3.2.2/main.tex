\textbf{Question:}\\
Draw a triangle $\triangle ABC$ in which $AB=5\,\text{cm}$, $BC=6\,\text{cm}$ and $\angle ABC=60^\circ$.\\[2pt]

\textbf{Solution:}\\
$\rightarrow$ Let
\begin{align}
a &= \|\vec{C}-\vec{B}\| = 6\,\text{cm} \tag{1}\\
b &= \|\vec{A}-\vec{C}\| \ \ (\text{to be found}) \tag{2}\\
c &= \|\vec{B}-\vec{A}\| = 5\,\text{cm} \tag{3}
\end{align}

$\rightarrow$ By cosine law in $\triangle ABC$ at $\vec{B}$,
\begin{align}
b^{2} &= a^{2}+c^{2}-2ac\cos B \tag{4}\\
      &= 6^{2}+5^{2}-2\cdot6\cdot5\cdot\cos60^\circ \tag{5}\\
      &= 36+25-60\cdot\frac12 \tag{6}\\
      &= 31 \ \Rightarrow\ b=\sqrt{31}\approx 5.57\,\text{cm} \tag{7}
\end{align}

$\rightarrow$ Choose coordinates (placing $B$ at the origin and $BC$ on the $x$-axis):
\begin{align}
\vec{B} &= \myvec{0\\0} \tag{8}\\
\vec{C} &= \myvec{6\\0} \tag{9}\\
\vec{A} &= c\,\myvec{\cos B\\ \sin B}
       = \myvec{\tfrac{5}{2}\\ \tfrac{5\sqrt{3}}{2}}
       \approx \myvec{2.5\\4.33} \tag{10}
\end{align}

$\rightarrow$ Therefore, the vertices of $\triangle ABC$ are:
\begin{align}
\vec{A} &= \myvec{2.5\\4.33}, \qquad
\vec{B} = \myvec{0\\0}, \qquad
\vec{C} = \myvec{6\\0} \tag{11}
\end{align}
