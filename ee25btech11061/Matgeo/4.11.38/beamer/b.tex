\begin{frame}{Question}
Find the area of the region bounded by the lines $y=2x+1$, $y=3x+1$ and $x=4$. \\
\end{frame}
\begin{frame}{Solution}
Given three lines are
\begin{align}
\myvec{-2 & 1}\myvec{x\\y} = 1 \Longrightarrow \vec{n}^\top\vec{x} = 1 \tag{1}
\end{align}
\begin{align}
\myvec{-3 & 1}\myvec{x\\y} = 1 \Longrightarrow \vec{m}^\top\vec{x} = 1 \tag{2}
\end{align}
\begin{align}
\myvec{1 & 0}\myvec{x\\y} = 4 \Longrightarrow \vec{p}^\top\vec{x} = 4 \tag{3}
\end{align}
  \end{frame}
  \begin{frame}{Solution}
  The three lines form a triangle.  
The vertices of triangle are obtained by 

\textbf{Intersection of : (1) and (3)}
\begin{align}
\myvec{-2 & 1 &\vrule& 1\\
1 & 0 &\vrule& 4} \xrightarrow{R_1 \longrightarrow R_1+2R_2}
\myvec{0 & 1 &\vrule& 9\\
1 & 0 &\vrule& 4}
\end{align}
From the second row we get $y=9$ so $x=4 \Longrightarrow
\vec{A} = \myvec{4\\9}$
\end{frame}

\begin{frame}{Solution}
\textbf{Intersection of : (2) and (3)}
\begin{align}
\myvec{-3 & 1 &\vrule& 1\\
1 & 0 &\vrule& 4} \xrightarrow{R_1 \longrightarrow R_1+3R_2}
\myvec{0 & 1 &\vrule& 13\\
1 & 0 &\vrule& 4}
\end{align}
From the second row we get $y=13$ so $x=4 \Longrightarrow
\vec{B} = \myvec{4\\13}$

\textbf{Intersection of : (1) and (2)}
\begin{align}
\myvec{-2 & 1 &\vrule& 1\\
-3 & 1 &\vrule& 1} \xrightarrow{R_2 \longrightarrow R_2-R_1}
\myvec{-2 & 1 &\vrule& 1\\
-1 & 0 &\vrule& 0}
\end{align}
From the second row we get $x=0$ so $y=1 \Longrightarrow
\vec{C} = \myvec{0\\1}$

\end{frame}

\begin{frame}{Solution}
\begin{align}
\vec{A}-\vec{B} = \myvec{0\\-4}, \quad \vec{A}-\vec{C} = \myvec{4\\8}   
\end{align}
\begin{align}
\norm{(\vec{A}-\vec{B})\times(\vec{A}-\vec{C})} 
= \absdet{\myvec{0 & -4\\ 4 & 8}} = |0-(-16)| = 16
\end{align}

\begin{align}
Area \ of \ the \ triangle 
= \frac{1}{2}{\norm{(\vec{A}-\vec{B})\times(\vec{A}-\vec{C})}}
= \frac{16}{2} = 8
\end{align}


Area of the triangle formed by the lines $y=2x+1$, $y=3x+1$ and $x=4$ is $8$ square units.


\end{frame}

#include <math.h>

static void intersect(double a1,double b1,double c1,
                      double a2,double b2,double c2,
                      double *x,double *y)
{
    double D = a1*b2 - a2*b1;
    *x = (b1*(-c2) - b2*(-c1)) / D;
    *y = (a2*(-c1) - a1*(-c2)) / D;
}

// Problem: y=2x+1, y=3x+1, x=4
// In ax+by+c=0 form: -2x + y -1 = 0 ; -3x + y -1 = 0 ; 1x + 0y -4 = 0

    double x1,y1,x2,y2,x3,y3;

    // L1 & L2
    intersect(-2, 1, -1,  -3, 1, -1,  &x1,&y1);
    // L2 & L3
    intersect(-3, 1, -1,   1, 0, -4,  &x2,&y2);
    // L3 & L1
    intersect( 1, 0, -4,  -2, 1, -1,  &x3,&y3);

    // Area = 1/2 * |det([B-A, C-A])|
    double ABx = x2 - x1, ABy = y2 - y1;
    double ACx = x3 - x1, ACy = y3 - y1;
    double area = 0.5 * fabs(ABx*ACy - ABy*ACx);
    return area;
}

import numpy as np
import matplotlib.pyplot as plt
import os

def intersection(a1,b1,c1, a2,b2,c2):
    A = np.array([[a1,b1],[a2,b2]], dtype=float)
    B = np.array([-c1,-c2], dtype=float)
    x, y = np.linalg.solve(A, B)
    return np.array([x, y])

L1 = (-2, 1, -1)
L2 = (-3, 1, -1)
L3 = (1, 0, -4)

A = intersection(*L1, *L2)
B = intersection(*L2, *L3)
C = intersection(*L1, *L3)
V = np.vstack([A,B,C])
