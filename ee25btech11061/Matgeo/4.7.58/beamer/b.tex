\begin{frame}{Question}
 Find the distance between the parallel lines $3x-4y+7=0$ and $3x-4y+5=0$.\\[8pt]
\end{frame}
\begin{frame}{Solution}
Let
\begin{align}
\vec{x} = \myvec{x\\y} \tag{1}
\end{align}

The given lines can be expressed in matrix form as
\begin{align}
\myvec{3 & -4}\vec{x} &= -7 \quad \text{and} \quad \myvec{3 & -4}\vec{x} = -5 \tag{2}
\end{align}

Here, the normal vector is
\begin{align}
\vec{n} = \myvec{3\\-4}, \quad c_1=-7, \quad c_2=-5 \tag{3}
\end{align}
  \end{frame}
  \begin{frame}{Solution}
  The distance between two parallel lines is
\begin{align}
d = \frac{|c_1-c_2|}{\|\vec{n}\|} \tag{4}
\end{align}


\begin{align}
\|\vec{n}\| &= \sqrt{3^2+(-4)^2} = \sqrt{9+16} = 5 \tag{5}\\
|c_1-c_2| &= |-7-(-5)| = |-2| = 2 \tag{6}
\end{align}

Therefore,
\begin{align}
d = \frac{2}{5} \tag{7}
\end{align}
\end{frame}

#include <math.h>

typedef struct {
    double distance;   // |c1 - c2| / sqrt(a^2 + b^2)
    double norm_n;     // sqrt(a^2 + b^2)
    double delta_c;    // |c1 - c2|
} DistOut;

// Distance between parallel lines: a x + b y + c1 = 0  and  a x + b y + c2 = 0
DistOut distance_parallel_pack(double a, double b, double c1, double c2) {
    DistOut out;
    out.norm_n  = sqrt(a*a + b*b);
    out.delta_c = fabs(c1 - c2);
    out.distance = (out.norm_n > 0.0) ? (out.delta_c / out.norm_n) : NAN;



\end{lstlisting}
