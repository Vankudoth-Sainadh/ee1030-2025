\begin{frame}{Question}
The equation of a line is $3x-4y+10=0$. Find its slope and the $x$ and $y$ intercepts. \\[6pt]
\end{frame}
\begin{frame}{Solution}
Let $\vec{x}=\myvec{x\\y}$. Then
\begin{align}
\myvec{3 & -4}\,\vec{x} &= -10 \tag{1}
\end{align}
Comparing with $\vec{n}^{T}\vec{x}=c$, we identify
\begin{align}
\vec{n}=\myvec{3\\-4}, \qquad c=-10. \tag{2}
\end{align}
A direction vector orthogonal to $\vec{n}$ is
\begin{align}
\vec{m}=\myvec{4\\3} \quad (\vec{n}^{T}\vec{m}=3\cdot4+(-4)\cdot3=0). \tag{3}
\end{align}
  \end{frame}
  \begin{frame}{Solution}
  Hence the slope is
\begin{align}
\text{slope}=\frac{m_y}{m_x}=\frac{3}{4}. \tag{4}
\end{align}

\textbf{x-intercept:} points on the $x$-axis have $\vec{x}=\myvec{x\\0}$. Substituting in (1),
\begin{align}
\myvec{3 & -4}\myvec{x\\0}=-10 \;\Rightarrow\; 3x=-10 \;\Rightarrow\; x=-\frac{10}{3}. \tag{5}
\end{align}

\begin{align}
\text{x-intercept}=\myvec{-\tfrac{10}{3}\\[2pt]0}. \tag{6}
\end{align}
\end{frame}

\begin{frame}{Solution}
\textbf{y-intercept:} points on the $y$-axis have $\vec{x}=\myvec{0\\y}$. Substituting in (1),
\begin{align}
\myvec{3 & -4}\myvec{0\\y}=-10 \;\Rightarrow\; -4y=-10 \;\Rightarrow\; y=\frac{5}{2}. \tag{7}
\end{align}

\begin{align}
\text{y-intercept}=\myvec{0\\[2pt]\tfrac{5}{2}}. \tag{8}
\end{align}
\end{frame}

#include <math.h>

// ax + by + c = 0  → slope = -a/b (if b ≠ 0)
// x-intercept = (-c/a, 0) if a ≠ 0
// y-intercept = (0, -c/b) if b ≠ 0
int line_params(double a, double b, double c,
                double* slope, int* slope_defined,
                double* xint,  int* x_defined,
                double* yint,  int* y_defined)
{
    if (b != 0.0) { *slope = -a/b; *slope_defined = 1; }
    else          { *slope = NAN;  *slope_defined = 0; }

    if (a != 0.0) { *xint = -c/a;  *x_defined = 1; }
    else          { *xint = NAN;   *x_defined = 0; }

int example_3m4p10(double* slope, int* slope_defined,

                   double* yint,  int* y_defined)
